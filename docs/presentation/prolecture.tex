  \begin{center}
  \footnotesize
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Работа} & \textbf{Что меряет} & \textbf{Чего не хватает} \\
    \midrule
    OVO-Bench & backward / real-time / forward & MCQ, нет аудио, перцепция \\
    ProactiveVideoQA & open-ended, метрика PAUC & простые вопросы «сообщи, когда X» \\

    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.8em}
  \begin{itemize}
    \item В лекции почти \textbf{нет перцептивной сложности}: чистый рендер, крупные формулы, чёткая речь.
    \item Зато есть логическая структура: посылка $\rightarrow$ вывод $\rightarrow$ следствие.
    \item Значит, ответить вовремя можно, только \lead{поняв содержание} — а не распознав объект.
  \end{itemize}

\begin{frame}
  \frametitle{Что ещё лежит в сэмпле}
  \framesubtitle{чем судят ответ и что в разметке измерено, а не заявлено}

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      {\bfseries В каждом окне}\\[0.4em]
      {\small
      \texttt{response} — эталонный ответ\\[0.2em]
      \texttt{rubric} — по чему судья ставит балл\\[0.2em]
      \texttt{trigger\_modality} — visual / speech / sound\\[0.2em]
      \texttt{depends\_on} — ссылка на предыдущее окно\\[0.5em]
      Рубрика — то, что заменяет MCQ. Из \texttt{q01}:
      \begin{itemize}\setlength{\itemsep}{0pt}
        \item[] «названо число 2.5»
        \item[] «если расчёт показан — он взвешенный, с 1/2, 1/4, 1/8, 1/8»
        \item[] \lead{«НЕ засчитывать 1.75 — это исходный код, а не переставленный»}
      \end{itemize}}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\bfseries \texttt{validation} — результаты замеров}\\[0.4em]
      {\small
      По флагу на зонд, каждый заполняется \emph{ответом модели}, а не мнением:\\[0.3em]
      \texttt{solvable} — решается ли вообще\\[0.2em]
      \texttt{blind\_solved} — решается без видео\\[0.2em]
      \texttt{textonly\_solved} — хватает транскрипта\\[0.2em]
      \texttt{visiononly\_solved} — хватает кадров\\[0.2em]
      \texttt{frozenframe\_solved} — хватает одного кадра\\[0.6em]
      Отсюда выводится \texttt{audio\_dependency}; мнение разметчика живёт отдельно,
      в \texttt{audio\_dependency\_claim}, и на выводы не влияет.\\[0.4em]
      \texttt{status}: \texttt{draft} $\rightarrow$ \texttt{validated} / \texttt{rejected}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}

  \framesubtitle{каждый сэмпл — точка в кубе (акт, отношение, уровень рассуждения)}

  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      {\bfseries A. Тип акта}\\[0.4em]
      {\small
      \texttt{alert} — сигнал\\
      \texttt{monitor} — состояние\\
      \texttt{ground} — локализация\\
      \texttt{narrate} — описание\\
      \texttt{predict} — предсказание\\
      \texttt{verify} — верификация\\
      \texttt{counterfact} — контрфактика}
    \end{column}
    \begin{column}{0.33\textwidth}
      {\bfseries B. Отношение к триггеру}\\[0.4em]
      {\small
      \texttt{pre} — доказательство \emph{после} вопроса: \lead{умение ждать}\\[0.3em]
      \texttt{at} — здесь и сейчас\\[0.3em]
      \texttt{post} — ответ по памяти\\[0.5em]
      \color{muted} Один вопрос в двух режимах даёт контролируемую пару.}
    \end{column}
    \begin{column}{0.33\textwidth}
      {\bfseries C. Уровень рассуждения}\\[0.4em]
      {\small
      \texttt{L1} перцепция — считывается\\[0.3em]
      \texttt{L2} интеграция — связать два момента\\[0.3em]
      \texttt{L3} вывод — провести рассуждение, которого в видео нет\\[0.5em]
      \color{muted} Критерий L3: blind-бейзлайн не решает.}
    \end{column}
  \end{columns}

  \vspace{0.9em}
  {\color{muted}\small Ось C — то, чего нет у OmniPro: там модели не различаются именно по ней.}
\end{frame}

\begin{frame}
  \frametitle{Принципы, из которых вырос дизайн}
  \framesubtitle{каждый пункт закрывает провал, на котором погорела одна из разобранных работ}

  \begin{itemize}
    \item \lead{Метки ставятся замером, а не декларацией.} \texttt{audio\_dependency}
          выводится из ответов зондов модальностей; мнение разметчика живёт
          в отдельном поле и на выводы не влияет.
    \item \lead{Генератор $\neq$ судья $\neq$ участник} — правило \emph{на этап замеров}.
          Сейчас пропозер, ревьюеры и координатор — одна и та же модель, а судью
          и участника мы ещё не запускали. Судьёй при этом может быть чистая
          текстовая LLM: она читает рубрику и ответ, а видео не смотрит.
    \item \lead{Открытые ответы с рубрикой, без MCQ.} Перекошенный ключ ответов ломает бенчмарк тихо.
    \item \lead{Обязательные бейзлайны-пустышки:} blind, text-only, vision-only, frozen-frame,
          «отвечать сразу», «отвечать в конце», «молчать».
    \item \lead{Никаких квот} вида «ровно N сэмплов на задачу»: сколько прошло валидацию, столько и есть.
    \item \lead{Контаминация измеряется, а не отрицается:} пилот на видео 2026 года,
          контрольное видео 2017 года — чтобы разрыв blind-бейзлайна увидеть числом.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Как это работает на живых сэмплах}
  \framesubtitle{одна и та же проверка вслепую, противоположные исходы}

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      {\color{accent}\bfseries q01 — проверку прошёл}\\[0.4em]
      {\small
      \emph{«Как только впервые появится таблица кодовых слов разной длины
      с вероятностями — посчитай, каким станет среднее число бит, если поменять
      местами „вверх“ и „влево“».}\\[0.5em]
      Вопрос на 250-й секунде, таблица — на 314-й; ответа 2.5 бита в видео нет.\\[0.5em]
      Полу-слепой ревьюер \emph{назвал} 2.5 — и засчитан не был: его ответ начинался
      с «точного числа назвать не могу», дальше условная догадка и ветка с 2.0.
      \lead{Угадана постановка, а не найден ответ.}}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{warm}\bfseries q08 — отвергнут}\\[0.4em]
      {\small
      \emph{«Скажи до того, как автор расскажет сам: к какому источнику
      вероятностей букв обратился Шеннон и почему у него нет проблемы
      таблицы частот».}\\[0.5em]
      До просмотра ревьюер выдал \textbf{обе} половины ответа: живой носитель языка
      и рост числа контекстов как $27^n$. Три пункта рубрики из трёх, без единого
      допущения о кадре.\\[0.5em]
      \lead{\texttt{blind\_solved = true}}, \texttt{status: rejected}; заодно уровень
      понижен \texttt{L3} $\rightarrow$ \texttt{L1}.}
    \end{column}
  \end{columns}

  \vspace{0.2em}
  {\color{muted}\footnotesize Правило: опора на общеизвестный факт — контаминация, догадка из постановки — нет.}
\end{frame}

\begin{frame}
  \frametitle{Три ревьюера, и каждый не знает о других}
  \framesubtitle{запускаются параллельно; файлы не правит никто из них}

  \begin{itemize}
    \item \lead{Полу-слепой.} Видит материал \emph{до} \texttt{t\_end}, но не знает
          \texttt{t\_start} — и сам называет момент, когда ответ стал возможен.
          Проверяющий, который знает ответ, «узнаёт» его на полукадре и раз за разом
          двигает левую границу окна влево; этого знания у него нет.
          \begin{itemize}
            \item[] {\small Отдельно записывает \texttt{prior\_answer} — свой ответ \textbf{до} просмотра,
                    дословно. Из него координатор и выводит \texttt{blind\_solved}.}
          \end{itemize}
    \item \lead{Ревьюер сэмпла.} Всё, что относится к вопросу как целому: формулировка,
          рубрика, приманки, эталонный ответ, ссылки между окнами. Границы окон — не его работа.
    \item \lead{Пооконный ревьюер, по одному на окно.} Роль отделена не для симметрии:
          у одного ревьюера на вопросе с девятью окнами бюджета кадров хватало
          на первое, а дальше он додумывал.
  \end{itemize}

  \vspace{0.5em}
  {\color{muted}\small Каждый возвращает структурированное ревью по контракту
  (\texttt{schema/review.py}), а правки применяет координатор — так видно, кто что нашёл,
  и повторный круг уходит свежим субагентам.}
\end{frame}

\begin{frame}
  \frametitle{Четыре зонда: метка получается замером}
  \framesubtitle{каждому дают ровно один канал и смотрят, ответит ли}

  \begin{center}
  \small
  \begin{tabular}{@{}llll@{}}
    \toprule
    \textbf{Зонд} & \textbf{Материал} & \textbf{Кадров} & \textbf{Заполняет} \\
    \midrule
    \texttt{text}   & транскрипт до \texttt{t\_end} & 0        & \texttt{textonly\_solved} \\
    \texttt{frame}  & один кадр в середине окна     & 1        & \texttt{frozenframe\_solved} \\
    \texttt{vision} & кадры до \texttt{t\_end}      & до 40    & \texttt{visiononly\_solved} \\
    \texttt{full}   & транскрипт и кадры            & до 40    & \texttt{solvable} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.6em}
  {\small
  \texttt{solvable = textonly OR visiononly OR} ответ \texttt{full}; \texttt{solvable = false}
  — \texttt{CRITICAL\_FAIL}, вопрос не решается и по полному материалу.\\[0.3em]
  \texttt{visiononly\_solved} вместе с \texttt{solvable} дают \lead{\texttt{audio\_dependency}}
  — руками эта метка не ставится.\\[0.3em]
  \texttt{frozenframe\_solved = true} при уровне \texttt{L2}/\texttt{L3} значит, что уровень завышен.}

  \vspace{0.5em}
  {\color{muted}\small Изоляция зонда держится на трёх вещах: промпт \emph{просит} не лезть
  в чужой канал, скрипт \emph{не даёт} смешать материал двух режимов в одной папке,
  а сам зонд обязан признаться в поле \texttt{violations}.}
\end{frame}

\begin{frame}
  \frametitle{Инструменты, которых не было}
  \framesubtitle{два места, где готовое не подошло}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Свой детектор смены сцен}\\[0.5em]
      {\small
      Готовые детекторы сравнивают кадр с предыдущим — а типовой переход manim
      это плавный переезд за 2--4 секунды, где соседние кадры почти не отличаются.\\[0.5em]
      Считаем \lead{накопленный дрейф} от последнего снятого кадра и снимаем,
      когда анимация улеглась.\\[0.5em]
      Один порог на домен: \num{13.5} / \num{12.9} / \num{10.1} кадра на минуту
      на роликах 2026 и 2017 годов.\\[0.3em]
      На контрольном участке: \num{18} состояний из 18 против \num{10} у равномерной сетки.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Плотное описание видеоряда}\\[0.5em]
      {\small
      Документ вроде субтитров, только вместо реплик — что происходит на экране.
      Разметчик читает текст вместо просмотра десятков кадров.\\[0.5em]
      Два кадра в секунду, окна по 5 секунд с перекрытием, строгий JSON по схеме
      (guided decoding на стороне сервера).\\[0.5em]
      Пилот: \num{3878} кадров, \num{485} фрагментов, \num{20} минут на двух A100,
      \num{1939} секунд из 1939 покрыты событиями.\\[0.3em]
      \color{muted} Субтитры описывающей модели не показывают: она описывает только кадры.
      Иначе описание нельзя было бы отдать зонду vision-only вместо кадров.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Метрика}
  \framesubtitle{спроектирована, но ещё не реализована: задел на фазу замеров}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{PAUC.} На каждом окне строится ломаная «время $\rightarrow$ качество
      накопленных ответов», качество $s \in \{0,1,2\}$ ставит LLM-судья по рубрике.

      \vspace{0.6em}
      Стартовая точка $(t_{\text{start}}, 0.5)$: молчание лучше уверенного неверного ответа.

      \vspace{0.6em}
      \lead{Следствие, которое отчитываем явно:} стратегия «молчать всегда» получает
      \num{0.25} от максимума. Это пол шкалы, а не ноль.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Обязательно рядом:}
      \begin{itemize}
        \item[] {\small
        \textbf{False-Alarm Rate} — ответы вне окон\\[0.25em]
        \textbf{Trap rate} — попадания в дистракторы, отдельно от FAR\\[0.25em]
        \textbf{Latency} — от $t_{\text{start}}$, распределением\\[0.25em]
        \textbf{Silence rate}, \textbf{Trigger Recall}\\[0.25em]
        \textbf{Responses-per-sample} — иначе спам-стратегия невидима}
      \end{itemize}
      \vspace{0.4em}
      {\color{muted}\small Агрегация по сэмплам, а не по окнам: иначе итог — заложник
      одного самого простого сэмпла с девятью триггерами.}
    \end{column}
  \end{columns}
\end{frame}
